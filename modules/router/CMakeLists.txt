cmake_minimum_required(VERSION 3.16)
project(model_router LANGUAGES CXX)

add_library(model_router model_router.cpp)
target_include_directories(model_router PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(model_router PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../..)
target_include_directories(model_router PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../ai-gateway/third_party)
target_sources(model_router PRIVATE contract_adapter.cpp)
target_compile_features(model_router PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(model_router PUBLIC /utf-8)
endif()

add_executable(router_demo demo.cpp)
target_link_libraries(router_demo PRIVATE model_router)
add_executable(router_tests tests.cpp)
target_link_libraries(router_tests PRIVATE model_router)
enable_testing()
add_test(NAME router_checks COMMAND router_tests)
add_executable(router_contract_tests contract_tests.cpp)
target_link_libraries(router_contract_tests PRIVATE model_router)
add_executable(router_contract_demo contract_demo.cpp)
target_link_libraries(router_contract_demo PRIVATE model_router)
add_test(NAME router_contract_checks COMMAND router_contract_tests
    ${CMAKE_CURRENT_SOURCE_DIR}/../../config/models.example.json)
