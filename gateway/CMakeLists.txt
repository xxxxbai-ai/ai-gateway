cmake_minimum_required(VERSION 3.20)
project(GatewayA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
find_package(Threads REQUIRED)
if(MSVC)
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
add_compile_definitions(NOMINMAX)
set(team_root "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(gateway_a STATIC service.cpp service.h)
target_include_directories(gateway_a PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${team_root}/shared")
target_include_directories(gateway_a SYSTEM PUBLIC "${team_root}/ai-gateway/third_party")
target_link_libraries(gateway_a PUBLIC Threads::Threads)

# 由组员在自己的构建文件中提供模块实现；本目录不复制或修改 B/C/D/E。
set(GATEWAY_MODULES_CMAKE "" CACHE FILEPATH "Optional team module CMake file")
set(GATEWAY_MODULE_SOURCES "" CACHE STRING "B/C/D implementation source files")
set(GATEWAY_MODULE_LIBRARIES "" CACHE STRING "B/C/D implementation libraries/targets")
if(GATEWAY_MODULES_CMAKE)
    include("${GATEWAY_MODULES_CMAKE}")
endif()
function(configure_gateway_host target)
    target_include_directories(${target} SYSTEM PRIVATE "${team_root}/ai-gateway/third_party/cpp-httplib")
    target_link_libraries(${target} PRIVATE gateway_a)
    if(WIN32)
        target_link_libraries(${target} PRIVATE ws2_32 shell32)
    endif()
endfunction()
if(GATEWAY_MODULE_SOURCES OR GATEWAY_MODULE_LIBRARIES)
    add_executable(gateway main.cpp ${GATEWAY_MODULE_SOURCES})
    configure_gateway_host(gateway)
    if(GATEWAY_MODULE_LIBRARIES)
        target_link_libraries(gateway PRIVATE ${GATEWAY_MODULE_LIBRARIES})
    endif()
else()
    message(STATUS "Building A and its tests only; gateway.exe requires the B/C/D public implementations.")
endif()

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_library(a_test_doubles STATIC tests/test_doubles.cpp)
    target_link_libraries(a_test_doubles PUBLIC gateway_a)
    add_executable(gateway_a_tests tests/service_tests.cpp)
    target_link_libraries(gateway_a_tests PRIVATE gateway_a a_test_doubles)
    if(WIN32)
        target_link_libraries(gateway_a_tests PRIVATE shell32)
    endif()
    add_executable(gateway_test_host main.cpp)
    target_compile_definitions(gateway_test_host PRIVATE A_GATEWAY_TEST_HOST=1)
    configure_gateway_host(gateway_test_host)
    target_link_libraries(gateway_test_host PRIVATE a_test_doubles)
    add_test(NAME a_service_contract COMMAND gateway_a_tests "${CMAKE_BINARY_DIR}/test-work/service")
    add_test(NAME a_http_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/http_tests.py"
        --gateway "$<TARGET_FILE:gateway_test_host>" --work "${CMAKE_BINARY_DIR}/test-work/http")
    set_tests_properties(a_service_contract a_http_contract PROPERTIES TIMEOUT 45)
endif()

if(WIN32 AND NOT MSVC)
    get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(runtime libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
        if(EXISTS "${compiler_bin}/${runtime}")
            file(MAKE_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
            configure_file("${compiler_bin}/${runtime}" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/${runtime}" COPYONLY)
        endif()
    endforeach()
endif()
