cmake_minimum_required(VERSION 3.21)
project(AIGatewayClient VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network)
qt_standard_project_setup()
add_library(client_core STATIC
    conversation.cpp conversation.h gateway_client.cpp gateway_client.h
    widgets.cpp widgets.h mainwindow.cpp mainwindow.h)
target_link_libraries(client_core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
target_include_directories(client_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(client_core PRIVATE $<$<CXX_COMPILER_ID:GNU>:-Wall;-Wextra;-Wpedantic>)
qt_add_executable(AIGatewayClient WIN32 main.cpp)
target_link_libraries(AIGatewayClient PRIVATE client_core)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/app.ico")
    target_sources(AIGatewayClient PRIVATE app.rc)
endif()
set_target_properties(AIGatewayClient PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(client_tests tests/client_tests.cpp)
    target_link_libraries(client_tests PRIVATE client_core Qt6::Test)
    set_target_properties(client_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" WIN32_EXECUTABLE FALSE)
    add_test(NAME client_delivery COMMAND client_tests -o "${CMAKE_BINARY_DIR}/client-tests.txt,txt")
    if(WIN32)
        set_tests_properties(client_delivery PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows" TIMEOUT 90)
    else()
        set_tests_properties(client_delivery PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
    endif()
endif()
